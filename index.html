function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!data.reportDate) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Missing reportDate"})) .setMimeType(ContentService.MimeType.JSON); } var reportDate = new Date(data.reportDate); var monthStr = (reportDate.getMonth() + 1) + "/" + reportDate.getFullYear(); function insertChronologically(sheet, dateColIndex, rowData, startRow) { startRow = startRow || 4; var lastRow = sheet.getLastRow(); if (lastRow < startRow) { sheet.getRange(startRow, 1, 1, rowData.length).setValues([rowData]); return; } var dates = sheet.getRange(startRow, dateColIndex, lastRow - startRow + 1, 1).getValues(); var inserted = false; for (var i = 0; i < dates.length; i++) { if (!dates[i][0]) continue; var currentDate = new Date(dates[i][0]); if (!isNaN(currentDate.getTime()) && reportDate < currentDate) { sheet.insertRowBefore(startRow + i); sheet.getRange(startRow + i, 1, 1, rowData.length).setValues([rowData]); inserted = true; break; } } if (!inserted) { sheet.appendRow(rowData); } } // 1. מעקב יציקות בטון if (data.pours && data.pours.length > 0) { var sheetPours = ss.getSheetByName("מעקב יציקות בטון"); if (sheetPours) { data.pours.forEach(function(p) { if (!p.volume) return; var rowData = [ data.reportDate, data.lot, p.building, p.floor, "", p.element, "", "", p.type === "CLSM" ? p.volume : "", p.type === "ב-20" ? p.volume : "", p.type === "ב-30" ? p.volume : "", p.type === "ב-40" ? p.volume : "", p.type === "ב-50" ? p.volume : "", p.type === "ב-60" ? p.volume : "", p.method, p.volume, "", "", p.testTaken || "לא", "", "", "", "", p.notes ]; insertChronologically(sheetPours, 1, rowData, 4); }); } } // 2. מעקב פועלי יומית if (data.dailyLabor && data.dailyLabor.hasLabor && data.dailyLabor.desc) { var sheetDaily = ss.getSheetByName("מעקב פועלי יומית"); if (sheetDaily) { var contractorNameValue = (data.dailyLabor.isCharge === "כן") ? data.dailyLabor.chargeName : ""; var rowDataDaily = [ data.lot, monthStr, data.reportDate, data.dailyLabor.count, data.dailyLabor.desc, data.dailyLabor.hours, data.dailyLabor.isCharge, contractorNameValue, data.dailyLabor.contractor ]; insertChronologically(sheetDaily, 3, rowDataDaily, 4); } } // 3. מעקב פועלי שלד+גמר var sheetSkeleton = ss.getSheetByName("מעקב פועלי שלד+גמר"); if (sheetSkeleton) { var daysNames = ["ראשון", "שני", "שלישי", "רביעי", "חמישי", "שישי", "שבת"]; var dayName = daysNames[reportDate.getDay()]; var skeletonDesc = ""; var finishingDesc = ""; var blocksDesc = ""; var generalColumnM = ""; // עמודה M עבור מלאכות נוספות / קבלני משנה במגרש 8 if (data.lot === "מגרש 12") { var skelArr = []; var blockArr = []; var finishArr = []; var buildings = ["121", "122", "123", "124"]; buildings.forEach(function(bld) { var s = data["b" + bld + "_skeleton"] || ""; var b = data["b" + bld + "_blocks"] || ""; var f = data["b" + bld + "_finishing"] || ""; if (s) skelArr.push("בניין " + bld + ": " + s); if (b) blockArr.push("בניין " + bld + ": " + b); if (f) finishArr.push("בניין " + bld + ": " + f); }); if (data.parkingWorks) { finishArr.push("חניונים ומחסנים: " + data.parkingWorks); } skeletonDesc = skelArr.join(" | "); blocksDesc = blockArr.join(" | "); finishingDesc = finishArr.join(" | "); } else { // עבור מגרש 8 skeletonDesc = data.genSkeletonWorks || ""; finishingDesc = data.genFinishingWorks || ""; generalColumnM = data.genSubcontractors || ""; } var rowDataSkeleton = [ data.lot, // א - מגרש monthStr, // ב - חודש data.reportDate, // ג - תאריך dayName, // ד - יום data.skeletonCount || 0, // ה - פועלי שלד 0, // ו - פועלים הודים skeletonDesc, // ז - תיאור שלד data.finishCount || 0, // ח - פועלי גמרים finishingDesc, // ט - תיאור גמר "", // י - לחיוב קבלן data.blockCount || 0, // יא - פועלי בלוקים blocksDesc, // יב - תיאור בלוקים generalColumnM // יג (עמודה M) - כללי (קבלני משנה / מלאכות נוספות למגרש 8) ]; insertChronologically(sheetSkeleton, 3, rowDataSkeleton, 4); } // 4. מעקב כלי עפר וציוד if (data.equipment && data.equipment.hasEquip && data.equipment.type) { var sheetEquip = ss.getSheetByName("מעקב כלי עפר"); if (sheetEquip) { var isLot8 = data.lot && data.lot.includes("8"); var dateColIndex = isLot8 ? 8 : 2; var startCol = isLot8 ? 7 : 1; var rowDataEquip = ["", data.reportDate, data.equipment.type, data.equipment.hours, data.equipment.notes]; var lastRow = sheetEquip.getLastRow(); var inserted = false; var startRow = 4; if (lastRow >= startRow) { var dates = sheetEquip.getRange(startRow, dateColIndex, lastRow - startRow + 1, 1).getValues(); for (var i = 0; i < dates.length; i++) { if (!dates[i][0]) continue; var currentDate = new Date(dates[i][0]); if (!isNaN(currentDate.getTime()) && reportDate < currentDate) { sheetEquip.insertRowBefore(startRow + i); sheetEquip.getRange(startRow + i, startCol, 1, rowDataEquip.length).setValues([rowDataEquip]); inserted = true; break; } } } if (!inserted) { var newRow = isLot8 ? ["", "", "", "", "", "", "", data.reportDate, data.equipment.type, data.equipment.hours, data.equipment.notes] : ["", data.reportDate, data.equipment.type, data.equipment.hours, data.equipment.notes]; sheetEquip.appendRow(newRow); } } } return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
